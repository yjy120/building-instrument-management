### 1. SecurityConfig — 安全配置

```java
@Configuration
@EnableWebSecurity
@EnableGlobalMethodSecurity(prePostEnabled = true)
public class SecurityConfig {

    @Bean
    public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
        http.csrf().disable()
            .sessionManagement().sessionCreationPolicy(SessionCreationPolicy.STATELESS)
            .and()
            .authorizeHttpRequests()
            .antMatchers("/api/auth/**").permitAll()
            .anyRequest().authenticated()
            .and()
            .addFilterBefore(jwtAuthenticationFilter(), UsernamePasswordAuthenticationFilter.class);
        return http.build();
    }

    @Bean public PasswordEncoder passwordEncoder() { return new BCryptPasswordEncoder(); }

    @Bean
    public AuthenticationManager authenticationManager(AuthenticationConfiguration c) throws Exception {
        return c.getAuthenticationManager();
    }

    // JWT 过滤器
    public static class JwtAuthenticationFilter extends OncePerRequestFilter {
        @Override
        protected void doFilterInternal(HttpServletRequest req, HttpServletResponse res, FilterChain chain)
                throws ServletException, IOException {
            String auth = req.getHeader("Authorization");
            if (auth != null && auth.startsWith("Bearer ") && jwtUtil.isTokenValid(auth.substring(7))) {
                String username = jwtUtil.extractUsername(auth.substring(7));
                var ud = userDetailsService.loadUserByUsername(username);
                SecurityContextHolder.getContext().setAuthentication(
                    new UsernamePasswordAuthenticationToken(ud, null, ud.getAuthorities()));
            }
            chain.doFilter(req, res);
        }
    }
}
```

### 2. JwtUtil — JWT 工具

```java
@Component
public class JwtUtil {
    @Value("${jwt.secret}") private String secret;
    @Value("${jwt.expiration}") private Long expiration;

    private SecretKey getSigningKey() {
        byte[] k = secret.getBytes(StandardCharsets.UTF_8);
        if (k.length < 32) { byte[] nk = new byte[32]; System.arraycopy(k, 0, nk, 0, k.length); return Keys.hmacShaKeyFor(nk); }
        return Keys.hmacShaKeyFor(k);
    }

    public String generateToken(String username) {
        return Jwts.builder().setSubject(username).setIssuedAt(new Date())
                .setExpiration(new Date(System.currentTimeMillis() + expiration))
                .signWith(getSigningKey()).compact();
    }

    public String extractUsername(String token) { return getClaims(token).getSubject(); }

    public boolean isTokenValid(String token) {
        try { return getClaims(token) != null && !getClaims(token).getExpiration().before(new Date()); }
        catch (Exception e) { return false; }
    }

    private Claims getClaims(String token) {
        return Jwts.parserBuilder().setSigningKey(getSigningKey()).build().parseClaimsJws(token).getBody();
    }
}
```

```java
@Service
public class AuthService {
    public LoginResponse login(LoginRequest request) {
        SysUser user = userRepository.findByUsername(request.getUsername())
                .orElseThrow(() -> new RuntimeException("用户不存在"));
        if (!passwordEncoder.matches(request.getPassword(), user.getPassword()))
            throw new RuntimeException("用户名或密码错误");
        String token = jwtUtil.generateToken(user.getUsername());
        return new LoginResponse(token, user.getId(), user.getUsername(),
                user.getRealName(), user.getRole(), user.getDepartment());
    }
}
```

### 4. DeviceService — 设备管理

```java
@Service
public class DeviceService {
    private static final BigDecimal LARGE_THRESHOLD = new BigDecimal("5000");

    public Device createDevice(DeviceRequest req) {
        if (deviceRepository.existsByDeviceCode(req.getDeviceCode()))
            throw new RuntimeException("仪器编码已存在");
        Device device = new Device();
        device.setDeviceCode(req.getDeviceCode());
        device.setDeviceName(req.getDeviceName());
        device.setSpecification(req.getSpecification());
        device.setPurchasePrice(req.getPurchasePrice());
        device.setStatus(req.getStatus() != null ? req.getStatus() : "IN_USE");
        device.setLocation(req.getLocation());
        device.setCategory(req.getCategory());
        // 采购价 >= 5000 → 大型设备
        device.setDeviceType(req.getPurchasePrice().compareTo(LARGE_THRESHOLD) >= 0 ? "LARGE" : "SMALL");
        Device saved = deviceRepository.save(device);
        // 同步创建库存记录
        Stock stock = new Stock(); stock.setDeviceId(saved.getId());
        stock.setQuantity(0); stock.setThreshold(10);
        stockRepository.save(stock);
        return saved;
    }

    public Device updateDevice(Long id, DeviceRequest req) {
        Device device = deviceRepository.findById(id).orElseThrow(() -> new RuntimeException("仪器不存在"));
        if (!device.getDeviceCode().equals(req.getDeviceCode()) && deviceRepository.existsByDeviceCode(req.getDeviceCode()))
            throw new RuntimeException("仪器编码已存在");
        // ... 字段赋值同 create
        device.setDeviceType(req.getPurchasePrice().compareTo(LARGE_THRESHOLD) >= 0 ? "LARGE" : "SMALL");
        return deviceRepository.save(device);
    }

    public List<Device> searchDevices(String code, String name, String category, String status) {
        return deviceRepository.findByConditions(code, name, category, status);
    }

    public void deleteDevice(Long id) { deviceRepository.deleteById(id); }
}
```

### 5. StockService — 库存管理

```java
@Service
public class StockService {
    public Stock adjustStock(Long deviceId, Integer delta, String reason, Long operatorId) {
        Stock stock = stockRepository.findByDeviceId(deviceId)
                .orElseThrow(() -> new RuntimeException("库存记录不存在"));
        int newQty = stock.getQuantity() + delta;
        if (newQty < 0) throw new RuntimeException("库存不足");
        stock.setQuantity(newQty);
        stockRepository.save(stock);
        // 记录流水
        StockRecord record = new StockRecord();
        record.setDeviceId(deviceId);
        record.setType(delta > 0 ? "IN" : "OUT");
        record.setQuantity(Math.abs(delta));
        record.setReason(reason); record.setOperatorId(operatorId);
        stockRecordRepository.save(record);
        return stock;
    }
}
```

### 7. DeviceIssueReportService — 异常申请（核心业务：报修/报废分流）

```java
@Service
public class DeviceIssueReportService {

    // 检测员提交异常申请（不区分报修/报废）
    public DeviceIssueReport createReport(DeviceIssueReportRequest req, Long applicantId) {
        Device device = deviceRepository.findById(req.getDeviceId())
                .orElseThrow(() -> new RuntimeException("设备不存在"));
        DeviceIssueReport report = new DeviceIssueReport();
        report.setDeviceId(req.getDeviceId()); report.setApplicantId(applicantId);
        report.setFaultDescription(req.getFaultDescription());
        report.setDeviceType(device.getDeviceType());
        report.setStatus("PENDING");
        return reportRepository.save(report);
    }

    // 管理员处理 → 小型设备走报修流程
    @Transactional
    public DeviceIssueReport handleAsRepair(Long reportId, Long handlerId) {
        DeviceIssueReport report = reportRepository.findById(reportId)
                .orElseThrow(() -> new RuntimeException("异常申请不存在"));
        if (!"PENDING".equals(report.getStatus())) throw new RuntimeException("该申请已处理");

        RepairOrder repair = new RepairOrder();
        repair.setDeviceId(report.getDeviceId()); repair.setApplicantId(report.getApplicantId());
        repair.setFaultDescription(report.getFaultDescription()); repair.setStatus("PENDING");
        RepairOrder saved = repairOrderRepository.save(repair);

        report.setStatus("PROCESSED"); report.setHandleType("REPAIR");
        report.setHandleId(saved.getId());
        return reportRepository.save(report);
    }

    // 管理员处理 → 大型设备走报废流程
    @Transactional
    public DeviceIssueReport handleAsScrap(Long reportId, Long handlerId) {
        DeviceIssueReport report = reportRepository.findById(reportId)
                .orElseThrow(() -> new RuntimeException("异常申请不存在"));
        if (!"PENDING".equals(report.getStatus())) throw new RuntimeException("该申请已处理");

        ScrapRecord scrap = new ScrapRecord();
        scrap.setDeviceId(report.getDeviceId()); scrap.setApplicantId(report.getApplicantId());
        scrap.setReason(report.getFaultDescription()); scrap.setStatus("PENDING");
        ScrapRecord saved = scrapRecordRepository.save(scrap);

        report.setStatus("PROCESSED"); report.setHandleType("SCRAP");
        report.setHandleId(saved.getId());
        return reportRepository.save(report);
    }
}
```

```java
// 报修 — 更新状态时联动设备状态
public RepairOrder updateRepairStatus(Long id, String status, String comment) {
    RepairOrder order = repairOrderRepository.findById(id).orElseThrow(...);
    order.setStatus(status); order.setRepairComment(comment);
    if ("COMPLETED".equals(status)) {
        Device d = deviceRepository.findById(order.getDeviceId()).orElse(null);
        if (d != null) { d.setStatus("IN_USE"); deviceRepository.save(d); }
    }
    return repairOrderRepository.save(order);
}

// 报废 — 审批通过时设备标记为 SCRAP
public ScrapRecord reviewScrap(Long id, String status, String comment, Long reviewerId) {
    ScrapRecord record = scrapRecordRepository.findById(id).orElseThrow(...);
    record.setStatus(status); record.setReviewComment(comment); record.setReviewerId(reviewerId);
    if ("APPROVED".equals(status)) {
        record.setScrapDate(new Date());
        Device d = deviceRepository.findById(record.getDeviceId()).orElse(null);
        if (d != null) { d.setStatus("SCRAP"); deviceRepository.save(d); }
    }
    return scrapRecordRepository.save(record);
}

// 校准 — 录入时同步设备下次校准日期
public CalibrationRecord createCalibration(CalibrationRequest req) {
    CalibrationRecord r = new CalibrationRecord();
    r.setDeviceId(req.getDeviceId()); r.setCalibrationDate(req.getCalibrationDate());
    r.setNextCalibrationDate(req.getNextCalibrationDate());
    r.setCalibrationOrganization(req.getCalibrationOrganization()); r.setResult(req.getResult());
    Device d = deviceRepository.findById(req.getDeviceId()).orElseThrow(...);
    d.setNextCalibrationDate(req.getNextCalibrationDate());
    deviceRepository.save(d);
    return calibrationRecordRepository.save(r);
}
```
